Service to resolve entitlements
@Service
public class EntitlementService {
  private final AppUserRepository users;
  public EntitlementService(AppUserRepository users) { this.users = users; }

  public Set<String> getPermissionsForUser(String username) {
    return users.findByUsername(username)
      .map(u -> u.getRoles().stream()
        .flatMap(r -> r.getPermissions().stream())
        .map(Permission::getName)
        .collect(Collectors.toCollection(LinkedHashSet::new)))
      .orElseGet(Set::of);
  }
}

REST: return entitlements after login
@RestController
@RequestMapping("/api/entitlements")
public class EntitlementController {
  private final EntitlementService entitlementService;
  public EntitlementController(EntitlementService s) { this.entitlementService = s; }

  @GetMapping
  public Map<String, Object> myEntitlements(Authentication auth) {
    String username = auth.getName();
    Set<String> perms = entitlementService.getPermissionsForUser(username);
    return Map.of("username", username, "permissions", perms);
  }
}

Security config (resource server + method security)
@EnableWebSecurity
@EnableMethodSecurity // for @PreAuthorize
@Configuration
public class SecurityConfig {

  @Bean
  SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
    http
      .authorizeHttpRequests(auth -> auth
        .requestMatchers("/actuator/**").permitAll()
        .requestMatchers(HttpMethod.GET, "/api/entitlements").authenticated()
        .anyRequest().authenticated()
      )
      .oauth2ResourceServer(oauth2 -> oauth2.jwt()); // validate JWT from ForgeRock

    return http.build();
  }

  // Optional: map JWT claim -> authorities if ForgeRock sends entitlements/roles
  @Bean
  Converter<Jwt, AbstractAuthenticationToken> jwtAuthConverter() {
    return jwt -> {
      Collection<GrantedAuthority> authorities = new ArrayList<>();
      // Example: claim "permissions": ["vendor_ReadOnly"]
      Object perms = jwt.getClaims().get("permissions");
      if (perms instanceof Collection<?> c) {
        c.forEach(p -> authorities.add(new SimpleGrantedAuthority(p.toString())));
      }
      // Example: roles -> add as ROLE_*
      Object roles = jwt.getClaims().get("roles");
      if (roles instanceof Collection<?> c) {
        c.forEach(r -> authorities.add(new SimpleGrantedAuthority("ROLE_" + r.toString())));
      }
      return new JwtAuthenticationToken(jwt, authorities, jwt.getClaimAsString("sub"));
    };
  }

  @Bean
  JwtDecoder jwtDecoder() {
    // Configure with your issuer/jwk-set-uri; placeholder for brevity
    return NimbusJwtDecoder.withJwkSetUri("https://forgerock.example/jwks").build();
  }
}

Entitlement Context
// src/context/EntitlementContext.tsx
import React, { createContext, useContext, useEffect, useMemo, useState } from "react";
import http from "../api/http";

type EntitlementState = {
  loaded: boolean;
  username?: string;
  permissions: string[];
  has: (perm: string) => boolean;
  any: (...perms: string[]) => boolean;
};

const Ctx = createContext<EntitlementState>({
  loaded: false,
  permissions: [],
  has: () => false,
  any: () => false,
});

export const EntitlementProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [loaded, setLoaded] = useState(false);
  const [username, setUsername] = useState<string | undefined>();
  const [permissions, setPermissions] = useState<string[]>([]);

  useEffect(() => {
    // call once right after login (or on app load if token exists)
    http.get("/entitlements")
      .then(res => {
        setUsername(res.data.username);
        setPermissions(res.data.permissions || []);
      })
      .catch(() => setPermissions([]))
      .finally(() => setLoaded(true));
  }, []);

  const value = useMemo(() => ({
    loaded,
    username,
    permissions,
    has: (perm: string) => permissions.includes(perm),
    any: (...perms: string[]) => perms.some(p => permissions.includes(p)),
  }), [loaded, username, permissions]);

  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
};

export const useEntitlements = () => useContext(Ctx);

import React, { createContext, useContext, useEffect, useState } from "react";
import http from "../api/http";

type EntitlementContextType = {
  permissions: string[];
  has: (perm: string) => boolean;
  any: (...perms: string[]) => boolean;

const EntitlementContext = createContext<EntitlementContextType>({
  permissions: [],
  has: () => false,
  any: () => false,

  return <EntitlementContext.Provider value={value}>{children}</EntitlementContext.Provider>;
};

package com.example.vendex.security;

import jakarta.servlet.http.HttpServletRequest;
import lombok.RequiredArgsConstructor;
import org.aspectj.lang.annotation.Aspect;
import org.aspectj.lang.annotation.Before;
import org.aspectj.lang.annotation.Pointcut;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;

import java.util.List;

@Aspect
@Component
@RequiredArgsConstructor
public class EntitlementAspect {

    private final EntitlementService entitlementService;
    private final HttpServletRequest request;

    @Pointcut("@annotation(requireEntitlement)")
    public void callAt(RequireEntitlement requireEntitlement) {}

    @Before("callAt(requireEntitlement)")
    public void checkEntitlement(RequireEntitlement requireEntitlement) {
        Authentication auth = SecurityContextHolder.getContext().getAuthentication();
        String username = auth.getName();

        List<String> userEntitlements = entitlementService.getEntitlementsForUser(username);

        if (!userEntitlements.contains(requireEntitlement.value())) {
            throw new SecurityException("User does not have entitlement: " + requireEntitlement.value());
        }
    }
}

3) Strategy A — entitlements come in the JWT

Ask your IAM team to include a claim, e.g.:

Map them to authorities:

@Configuration
class JwtAuthoritiesMappingConfig {

  @Bean
  Converter<Jwt, AbstractAuthenticationToken> jwtAuthenticationConverter() {
    return jwt -> {
      List<GrantedAuthority> auths = new ArrayList<>();

      // map "permissions" claim to authorities exactly
      Object perms = jwt.getClaims().get("permissions");
      if (perms instanceof Collection<?> c) {
        c.forEach(p -> auths.add(new SimpleGrantedAuthority(p.toString())));
      }

      // optional: map "roles" to ROLE_ authorities
      Object roles = jwt.getClaims().get("roles");
      if (roles instanceof Collection<?> c) {
        c.forEach(r -> auths.add(new SimpleGrantedAuthority("ROLE_" + r.toString())));
      }

      String name = jwt.getClaimAsString("preferred_username");
      if (name == null) name = jwt.getSubject();

      return new JwtAuthenticationToken(jwt, auths, name);
    };
  }
}

4) Strategy B — entitlements from DB per request (your stated plan)

Here, JWT only identifies the user. We augment authorities by looking up DB entitlements once per request (or cache them).

4.1 A small filter to inject authorities
@Component
@RequiredArgsConstructor
class DbEntitlementsAugmentFilter extends OncePerRequestFilter {

  private final EntitlementService entitlementService;

  @Override
  protected void doFilterInternal(HttpServletRequest req, HttpServletResponse res, FilterChain chain)
      throws ServletException, IOException {

    Authentication auth = SecurityContextHolder.getContext().getAuthentication();

    // Only act for an authenticated JWT
    if (auth instanceof JwtAuthenticationToken jwtAuth && auth.isAuthenticated()) {
      String username = jwtAuth.getName(); // set by resource server; preferred_username or sub

      // Load entitlements from DB (could cache by username for 1-5 mins)
      Set<String> perms = entitlementService.byUsername(username);

      // Combine existing authorities (if any) with DB entitlements
      List<GrantedAuthority> merged = new ArrayList<>(jwtAuth.getAuthorities());
      for (String p : perms) merged.add(new SimpleGrantedAuthority(p));

      Authentication newAuth =
          new JwtAuthenticationToken(jwtAuth.getToken(), merged, username);

      SecurityContextHolder.getContext().setAuthentication(newAuth);
    }
    chain.doFilter(req, res);
  }
}


Register the filter after JWT auth:

@Configuration
class FilterRegistrationConfig {

  private final DbEntitlementsAugmentFilter dbFilter;

  FilterRegistrationConfig(DbEntitlementsAugmentFilter dbFilter){ this.dbFilter = dbFilter; }

  @Bean
  SecurityFilterChain chain(HttpSecurity http) throws Exception {
    http
      .csrf(csrf -> csrf.disable())
      .authorizeHttpRequests(auth -> auth
        .requestMatchers(HttpMethod.GET, "/api/entitlements").authenticated()
        .requestMatchers("/api/**").authenticated()
        .anyRequest().permitAll())
      .oauth2ResourceServer(oauth -> oauth.jwt());

    // Ensure our filter runs after Jwt is established
    http.addFilterAfter(dbFilter, org.springframework.security.oauth2.server.resource.web.BearerTokenAuthenticationFilter.class);
    return http.build();
  }
}
